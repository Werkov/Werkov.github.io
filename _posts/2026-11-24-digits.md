  <script>
    const svg = document.getElementById("graph");
    const edgesLayer = document.getElementById("edges-layer");
    const hud = document.getElementById("hud");
    const nodeMap = new Map();
    const adj = new Map();
    const incident = new Map();

    svg.querySelectorAll(".node").forEach(el => {
      const id = el.dataset.id;
      nodeMap.set(id, el);
      adj.set(id, []);
      incident.set(id, []);
    });

    svg.querySelectorAll(".edge").forEach(el => {
      const u = el.dataset.u, v = el.dataset.v;
      adj.get(u).push(v);
      adj.get(v).push(u);
      incident.get(u).push(el);
      incident.get(v).push(el);
    });

    let pinned = null;

    function clearFocus() {
      svg.classList.remove("has-focus");
      svg.querySelectorAll(".active, .neighbor, .incident").forEach(el => {
        el.classList.remove("active", "neighbor", "incident");
      });
      hud.textContent = "Hover any shape to highlight adjacent moves • Click to pin";
    }

    function focusNode(id) {
      clearFocus();
      if (!nodeMap.has(id)) return;
      svg.classList.add("has-focus");
      
      const activeEl = nodeMap.get(id);
      activeEl.classList.add("active");
      activeEl.parentNode.appendChild(activeEl);

      const nbrs = adj.get(id);
      nbrs.forEach(nid => {
          const nEl = nodeMap.get(nid);
          nEl.classList.add("neighbor");
          nEl.parentNode.appendChild(nEl);
      });
      
      incident.get(id).forEach(edgeEl => {
        edgeEl.classList.add("incident");
        edgesLayer.appendChild(edgeEl);
      });

      const pinNote = pinned === id ? " [PINNED — click bg to clear]" : "";
      const deg = activeEl.dataset.deg;
      hud.textContent = `${deg} single-stick moves available${pinNote}`;
    }

    nodeMap.forEach((el, id) => {
      el.addEventListener("mouseenter", () => { if (!pinned) focusNode(id); });
      el.addEventListener("mouseleave", () => { if (!pinned) clearFocus(); });
      el.addEventListener("click", e => {
        e.stopPropagation();
        pinned = pinned === id ? null : id;
        pinned ? focusNode(id) : clearFocus();
      });
    });

    svg.addEventListener("click", () => {
      pinned = null;
      clearFocus();
    });
  </script>
